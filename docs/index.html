1. 快速上手与插件规范

单脚本 scripts/<id>/

  • 入口：plugins/scripts/<id>/index.js
  • 配置：同级 config.json（便携独立）
  • 特色：在线热改、即存即生效、支持导入导出纯 JS
  • 适合：小游戏、趣味群聊互动、轻量小指令

完整应用 apps/<id>/

  • 入口：plugins/apps/<id>/index.js
  • 清单：具备 package.json 与 schema.json
  • 特色：ESM 运行时自愈、影子目录热重载、Zip 导出
  • 适合：表情包引擎、服务总线提供者、企业级业务

最小起步代码

在 plugins/scripts/hello/index.js 中写入：

export const name = '问候示例'
export const version = '1.0.0'

export default function setup(ctx) {
  ctx.logger.info('问候插件已初始化！')

  // 响应指定指令或前缀
  ctx.registerCommand(/^#?(打招呼|hello)$/i, async (e) => {
    // 跨协议友好回复发送者
    await e.reply({
      text: `你好呀，${e.sender.nickname || e.userId}！欢迎使用 Chaite。`,
      at: true,
    })
    return true // 返回 true 表示拦截并完成事件消费
  })
}

2. 事件系统与协议直通

框架将 OneBot、飞书等异构协议的消息与通知统一抽象为标准 ChaiteEvent，业务层彻底告别直接操作底层 CQ 码或散落的全局对象。

属性 / 方法 类型 说明
e.userId / e.groupId number | string | null 发送者用户 ID / 群号（私聊环境群号为 null）
e.isGroup / e.isPrivate boolean 当前会话场景判断布尔标记
e.text / e.rawText string e.text 已剥离机器人的 @ 与指令前缀，rawText 为纯原始文本
e.command / e.args string e.command 为命中的指令名；e.args 为指令后面的纯入参文本（已自动 trim）
e.eventType string 事件类别：message、poke、member_join、member_leave、request、recall
e.sender SenderInfo 发送者元数据：{ nickname, card, role, level }
e.isPoke / e.isSelfPoke boolean 是否为戳一戳通知，isSelfPoke 指示是否戳的是机器人自身
await e.reply(content) Promise<any> 跨协议多功能回复（支持纯文本、艾特、图片数组、引用或无引用）
await e.approve(remark?) / e.reject(reason?) Promise<any> 一键快速审批加好友或加群请求（仅在 request 申请事件中生效）
await e.deleteMsg(msgId?) Promise<any> 快捷撤回指定消息（若未传参则默认撤回当前事件所在消息）
await e.getMemberList() Promise<any[]> 群聊场景原生查询完整群成员列表
await e.callAction(action, params) Promise<any> 直通底层协议原生动作（如发送合并转发 send_group_forward_msg 等）

全生命周期事件监听范例

export default function setup(ctx) {
  // 1. 便捷指令入参读取 (e.args 自动剥除指令头)
  ctx.registerCommand('#echo', async (e) => {
    if (!e.args) return e.reply('请输入要复读的内容')
    await e.reply(`复读: ${e.args}`)
    return true
  })

  // 2. 群成员变动感知 (入群欢迎 & 离群播报)
  ctx.onMemberJoin(async (e) => {
    await e.reply({ text: `🎉 欢迎新群友 [@${e.userId}] 加入本群！`, at: e.userId })
    return true
  })
  ctx.onMemberLeave(async (e) => {
    ctx.logger.info(`群成员 ${e.userId} 已离开群聊 ${e.groupId} (原因: ${e.subType})`)
  })

  // 3. 申请审核自动化 (自动同意指定暗号申请)
  ctx.onRequest(async (e) => {
    if (e.comment?.includes('Chaite开发者')) {
      await e.approve('欢迎加入 Chaite 开发者交流圈')
      return true
    }
  })

  // 4. 撤回感知与防撤回留证
  ctx.onRecall(async (e) => {
    ctx.logger.warn(`检测到消息撤回: 消息ID=${e.messageId}, 操作人=${e.operatorId}, 原作者=${e.userId}`)
  })

  // 5. 戳一戳响应
  ctx.onPoke(async (e) => {
    if (e.isSelfPoke) {
      await e.reply('别戳我啦，再戳要生气啦！(ง •_•)ง')
      return true
    }
  })
}

3. 声明式配置与控制

每个插件的配置存放在其独立目录下的 config.json 中，业务代码通过 ctx.config 读取。放置同级 schema.json 即可由 Web 管理面板生成可视化配置表单。

[
  { "key": "apiKey", "label": "API 密钥", "type": "string" },
  { "key": "maxCount", "label": "并发上限", "type": "number", "default": 5 },
  {
    "key": "installEngine",
    "label": "原生引擎管理",
    "type": "button",
    "buttonText": "📦 一键安装引擎核心",
    "buttonType": "primary",
    "action": "installEngine",
    "description": "点击后由系统自动在后台执行依赖安装，免去命令行操作"
  }
]

字段类型： string, number, boolean, textarea, select, button（声明式交互按钮）。
按钮动作（Actions）： 在插件代码中通过 ctx.registerAction('installEngine', async () => { ... }) 处理点击响应。
控制台分栏： 面板自动将表单隔离为 🔌 插件配置（专有选项）与 ⚙️ 通用配置（独立触发前缀、群聊及用户黑白名单），杜绝杂糅。

4. AI 赋能与对话流

当预设性格活泼或有特定系统提示词，而插件需要执行提取、严谨计算等确定性任务时，可自由选择预设直通或纯净基元模式：

ctx.registerCommand('#翻译', async (e) => {
  // 方式 1: 走已配置的听话预设 (presetId)
  const res = await e.queryAI(e.text, { presetId: 'translator' })
  await e.reply(res)
  return true
})

ctx.registerCommand('#提取', async (e) => {
  // 方式 2: 纯净基元模式 (pure: true)，零人设干扰，直连渠道模型
  const jsonStr = await e.queryAI(e.text, {
    pure: true,
    channelId: 'my-openai-channel',
    systemPrompt: '你是一个严谨的数据提取助手，只返回标准 JSON 数据',
    temperature: 0.1,
  })
  await e.reply(jsonStr)
  return true
})

ctx.registerCommand('#总结讨论', async (e) => {
  if (!e.isGroup) return true
  await e.reply('正在阅读刚才的发言，请稍候...')
  // 携带群聊双水位线滑动窗口记录提问
  const summary = await e.askWithContext('请总结群友刚才的核心讨论，列出要点')
  await e.reply(summary)
  return true
})

5. 多媒体 Ref 体系规范 (Media Ref Specification)

为了防止多媒体体积撑爆模型上下文和网关负载，Chaite 统一采用 Ref 引用体系 (mediaRefStore) 对异构多媒体做轻量占位，支持大模型与插件按需延迟解析。

媒体类别 文本占位符格式 Ref 标识 核心能力与解析规范
图片 (Image) [图片 ref:img_xxxx] img_ 内置双水位线防爆与 GIF 三级动态压缩；支持 mediaRefStore.resolveImage(ref) 提取。
聊天记录 (Forward) [聊天记录:共 N 条记录 摘要:... ref:fwd_xxxx] fwd_ 预存图文预览摘要，无需大模型读取庞大原始 JSON 即可精确掌握会话脉络。
视频 (Video) [视频:xxx.mp4 ref:vid_xxxx] vid_ 携带文件名、后缀与大小；支持 mediaRefStore.resolveMedia(ref) 异步解析下载直链。
语音 (Voice) [语音 (5秒) ref:voice_xxxx] voice_ 包含精确秒数与音频文件路径，便于对接外部 STT 语音识别服务。
文件 (File) [文件:文档.pdf (3.5MB) ref:file_xxxx] file_ 包含群文件 ID；支持 mediaRefStore.resolveMedia(ref) 获取群文件直链下载地址。

提取群文件下载直链与多媒体按需解析范例

import { mediaRefStore } from 'chaite' // 或直接读取 globalThis.mediaRefStore

export default function setup(ctx) {
  ctx.registerCommand('#下载直链', async (e) => {
    // 匹配消息或回复中的文件 Ref: [文件:xxx.pdf (3.5MB) ref:file_xxxx]
    const match = e.rawText.match(/ref:(file_[a-z0-9_]+)/)
    if (!match) return e.reply('请引用或发送文件卡片并附带 #下载直链 指令')

    const fileRef = match[1]
    const item = mediaRefStore.get(fileRef)
    if (!item) return e.reply('文件引用已过期或不存在')

    // 自动调用底层协议获取群文件或私聊文件的下载直链
    const resolved = await mediaRefStore.resolveMedia(fileRef, globalThis.onebotClient)
    if (resolved?.url) {
      await e.reply(`📄 文件名: ${item.name}\n大小: ${item.fileSize ? Math.round(item.fileSize / 1024) + ' KB' : '未知'}\n直链地址: ${resolved.url}`)
    } else {
      await e.reply('直链解析失败，该文件可能已过期。')
    }
    return true
  })
}

6. 服务总线与原生渲染 (Shotium)

Chaite 提倡 Service Provider（服务提供者模式）：业务插件（如 meme、卡片仪表盘）只依赖抽象的 ctx.render({ html, url })，底层渲染引擎由独立的 service-renderer 插件负责承载与切换。

Shotium 本地原生引擎核心优势 (@pixel.js/shotium)

  • 无端口、无微服务：纯本地 Node-API 进程内驱动，无需配置任何 HTTP 端口或 Docker 容器；
  • 剥离臃肿：深度移除 V8 引擎与 DevTools，仅保留 Blink 排版与 Skia 绘图，内存占用极低；
  • 秒级冷启动：毫秒级输出 PNG、JPEG 或 WebP 图像流，彻底告别 Chromium 僵尸进程；
  • 免终端一键安装：在面板点击 📦 一键安装 Shotium 即可走国内镜像源自动装配完毕。
// 业务插件中使用渲染服务极简范例：
ctx.registerCommand('#生成卡片', async (e) => {
  const html = `
    <div style="background:#0f172a;color:#fff;padding:24px;border-radius:12px;font-family:sans-serif;">
      <h2 style="color:#38bdf8;margin:0 0 8px;">🚀 智能体状态卡片</h2>
      <p>用户: ${e.sender.nickname || e.userId}</p>
      <p>生成时间: ${new Date().toLocaleString()}</p>
    </div>
  `
  // 直接调用渲染总线，无论底层是 Shotium 还是 Puppeteer 均完美出图
  const imageBuf = await ctx.render({ html, format: 'png' })
  const base64 = Buffer.isBuffer(imageBuf) ? imageBuf.toString('base64') : String(imageBuf)
  await e.reply({
    text: '渲染完成：',
    images: [`base64://${base64}`]
  })
  return true
})

通用服务总线 (Service Provider Pattern)

Chaite 提供了轻量级的微内核服务总线机制：插件不仅可以使用现有的渲染服务，还可以向系统暴露通用服务，实现跨插件解耦调用。 

// 1. 服务提供方 (如 plugins/apps/service-translate/index.js)
ctx.registerService('translator', {
  async translate(text, targetLang = 'en') {
    // 执行本地模型或远程翻译接口
    return `[${targetLang}] ${text}`
  }
})

// 2. 服务消费方 (在其他任意插件中无感调用)
const translator = ctx.getService('translator')
if (translator) {
  const result = await translator.translate('你好世界')
}

7. 存储、工具与时限管理

轻量持久化 KV 存储

// 写入数据（自动 JSON 序列化存储至插件私有命名空间）
await ctx.storage.set('user_level_' + e.userId, 5)

// 获取数据（支持指定默认回退值）
const level = await ctx.storage.get('user_level_' + e.userId, 1)

// 删除条目
await ctx.storage.delete('user_level_' + e.userId)

Tool 工具注入与定时任务

export default function setup(ctx) {
  // 1. 注册 Function Calling 工具给预设模型使用
  ctx.registerTool({
    name: 'query_inventory',
    description: '查询某样物品的当前库存数量',
    parameters: {
      type: 'object',
      properties: {
        itemName: { type: 'string', description: '物品名称' }
      },
      required: ['itemName']
    }
  }, async (args) => {
    return { stock: 42, item: args.itemName }
  })

  // 2. 定时任务调度（插件重载或卸载时自动安全注销 Job）
  ctx.cron('0 8 * * *', async () => {
    ctx.logger.info('早上 8:00 定时广播触发！')
  })
}

系统内置管理与运维指令速查

指令前缀 功能说明 权限要求
#chaite / #帮助 呼出系统基础指令帮助列表 全体用户
#重置对话 / #清空上下文 清空当前用户在当前群/私聊会话的预设短期对话记忆 全体用户
#清空全部会话 / #resetall 强制清空数据库中所有用户和群聊的全部上下文 管理员 (Master)
#预设列表 / #查看预设 查看当前已加载的所有人设预设及其触发前缀 全体用户

#拉黑 @群友 [时长] 支持时限（如 10m、2h、1d），不填时长则为永久加入黑名单 管理员 (Master)
#解黑 @群友 解除用户的全局黑名单与时限黑名单 管理员 (Master)
#拉黑群 [群号] [时长] 支持时限临时拉黑指定群聊或永久入黑 管理员 (Master)
#解黑群 [群号] 解除群聊的黑名单限制 管理员 (Master)
