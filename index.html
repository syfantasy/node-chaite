1. 目录结构规范

单文件 scripts/<id>/

  • 入口：runner/plugins/scripts/<id>/index.js
  • 配置：同级 config.json（便携独立）
  • 模式：在线代码编辑、支持直接导入 .js 或带配置的 .zip 压缩包
  • 适合：小游戏、趣味互动、单指令脚本

工程包 apps/<id>/

  • 入口：runner/plugins/apps/<id>/index.js
  • 清单：package.json，同级 config.json
  • 模式：多文件完整模块，支持一键打包与导入 .zip
  • 适合：多功能套件、企业接口对接、复杂业务

2. 最小起步实例

在 runner/plugins/scripts/hello/index.js 创建脚本：

export const name = '打招呼插件'
export const version = '1.0.0'

export default function setup(ctx) {
  ctx.logger.info('插件启动成功！')

  // 读取插件专属配置
  const prefix = ctx.config.prefix || '#打招呼'

  // 注册指令
  ctx.registerCommand(prefix, async (e) => {
    await e.reply(`你好呀，${e.sender.nickname || e.userId}！`)
    return true // 返回 true 表示拦截并完成处理
  })
}

3. 统一事件对象 (ChaiteEvent) API 规范

字段 / 方法 类型 说明
e.userId string | number 发送者用户 ID（QQ 号等）
e.groupId number | null 群号（私聊时为 null）
e.isGroup / e.isPrivate boolean 当前是否为群聊 / 私聊环境
e.text string 清洗后的文本（已剥离 @ 机器人与指令前缀）
e.rawText string 未经清洗的原始消息文本
e.sender SenderInfo 发送者信息：{ nickname, card, role, title, level }
e.images ImageAttachment[] 携带的图片数组 [{ ref, url, mimeType }]
await e.reply(content) Promise<any> 统一回复方法（支持文本、{ text, at: true, images } 或 CQ 码自动转换）
await e.queryAI(prompt, [opt]) Promise<string> 直接调用系统配置的主模型/预设提问（单轮直通）
await e.askWithContext(prompt, [opt]) Promise<string> 携带群聊近期双水位线滑动窗口记录向模型提问

跨协议回复范例 (e.reply)

为避免将插件死绑定在 QQ 专有的 CQ 码上，推荐使用跨协议的结构化调用方式（框架会自动向 OneBot / 飞书转换对应原生消息段）：

// 1. 最简文本回复（默认带原消息引用）
await e.reply('处理成功喵！')

// 2. 不带引用回复 (独立单发)
await e.reply({
  text: '这是不带引用的独立回复消息',
  reply: false, // 设为 false 则不引用原消息
})

// 3. 结构化艾特回复（跨平台通用，无需手拼 CQ 码）
await e.reply({
  text: '这是你的查询结果',
  at: true,     // 自动 @ 发送者；亦可传入指定 QQ 号如 at: 123456
  reply: false, // 可自由组合是否带引用
})

// 4. 图文混合回复
await e.reply({
  text: '图表已生成：',
  images: ['https://example.com/chart.png'],
})

// 5. CQ 码兼容说明
// 框架底层已内置 CQ 码解析器，手写的 [CQ:at,qq=123] 或 [CQ:image,file=...] 也会自动转为原生 Segment。

4. 插件配置与 Schema 声明

每个插件的配置均保存在其独立目录下的 config.json，在代码中直接通过 ctx.config 访问。如需 Web 后台渲染可视化配置表单，同级放置 schema.json：

[
  { "key": "apiKey", "label": "第三方 API 密钥", "type": "string" },
  { "key": "prefix", "label": "自定义指令唤醒词", "type": "string", "default": "#打招呼" },
  { "key": "enableAt", "label": "回复时是否艾特", "type": "boolean", "default": true }
]

支持字段类型：string, number, boolean, textarea, select。

5. AI 能力赋能 (预设模式 vs 纯净基元模式)

当默认预设性格活泼/不听话，而插件需要执行提取、摘要、严谨计算等任务时，可自由选择以下方式：

ctx.registerCommand('#翻译', async (e) => {
  // 方式 1: 指定听话预设 ID (预设已绑定好渠道与模型，无需额外配置)
  const res = await e.queryAI(e.text, { presetId: 'translator' })
  await e.reply(res)
  return true
})

ctx.registerCommand('#提取', async (e) => {
  // 方式 2: 纯净基元对话 (完全不带预设性格与提示词，直接调渠道模型)
  const jsonStr = await e.queryAI(e.text, {
    pure: true,                     // 纯净模式，零预设污染
    channelId: 'my-openai-channel', // 可选：指定特定渠道 ID
    systemPrompt: '你是一个严格的数据提取助手，仅返回格式化后的 JSON',
    temperature: 0.1
  })
  await e.reply(jsonStr)
  return true
})

ctx.registerCommand('#总结讨论', async (e) => {
  if (!e.isGroup) return true
  await e.reply('正在阅读刚才的发言，请稍候...')
  // 携带群聊上下文提炼
  const summary = await e.askWithContext('请总结刚才群友的核心讨论，列出3个看点')
  await e.reply(summary)
  return true
})

6. 多媒体与资源引用体系 (Media Ref Specification)

在聊天场景中，图片、视频、语音、文件及合并转发消息体量庞大且异构。Chaite 统一采用 Ref 引用体系 (mediaRefStore) 对非纯文本媒体进行轻量化占位抽象，避免上下文和网关 Payload 爆破，同时允许插件和大模型按需延迟解析。

媒体类别 文本占位符格式 Ref 前缀 核心元数据与扩展能力
图片 (Image) [图片 ref:img_xxxx] img_ 包含图片直链/本地文件、Base64 缓存与 MIME。具备双水位线防爆与 GIF 三级动态压缩。支持 mediaRefStore.resolveImage(ref)。
聊天记录 (Forward) [聊天记录:共 N 条记录 摘要:... ref:fwd_xxxx] fwd_ 包含合并转发 ID (forwardId) 与前 10 条消息图文摘要预览 (previewMessages)，大模型无需读取庞大原始 JSON 即可理解上下文对话脉络。
视频 (Video) [视频:xxx.mp4 ref:vid_xxxx] vid_ 包含视频名称、后缀 (ext)、字节大小 (fileSize)。支持 mediaRefStore.resolveMedia(ref) 异步解析临时下载直链与本地缓存。
语音 (Voice) [语音 (5秒) ref:voice_xxxx] voice_ 包含语音精确时长 (duration) 及 Silk/Amr 资源路径，插件可结合外部语音识别服务进行 STT 文本转录。
文件 (File) [文件:文档.pdf (3.5MB) ref:file_xxxx] file_ 包含文件名、大小与群文件 ID (fileId)。支持 mediaRefStore.resolveMedia(ref) 自动获取群文件直链下载地址。

插件中提取与解析 Ref 示例

// 全系列媒体 Ref 提取与按需解析示例
import { mediaRefStore } from 'chaite' // 或直接读取 globalThis.mediaRefStore

export default function setup(ctx) {
  ctx.on('message', async (e) => {
    const raw = e.rawText || ''

    // 匹配消息中包含的任意多媒体 Ref
    // 覆盖格式：[视频:... ref:vid_xxx]、[文件:... ref:file_xxx]、[语音 ref:voice_xxx]、[聊天记录 ref:fwd_xxx]、[图片 ref:img_xxx]
    const refMatch = raw.match(/\[(?:视频|文件|语音|聊天记录|图片)[^\]]*ref:([a-z0-9_]+)\]/)
    if (!refMatch) return

    const ref = refMatch[1]
    const item = mediaRefStore.get(ref)
    if (!item) return

    switch (item.type) {
      case 'file': {
        // 自动解析群文件/私聊文件的下载直链或本地缓存
        const resolved = await mediaRefStore.resolveMedia(ref, globalThis.onebotClient)
        if (resolved?.url) {
          await e.reply(`已捕获文件: ${item.name} (${item.fileSize ? Math.round(item.fileSize / 1024) + 'KB' : '未知大小'})\n直链解析成功！`)
        }
        break
      }
      case 'video': {
        const resolved = await mediaRefStore.resolveMedia(ref, globalThis.onebotClient)
        console.log(`收到视频: ${item.name}，格式: ${item.ext}，直链: ${resolved?.url || '本地缓存'}`)
        break
      }
      case 'voice': {
        console.log(`收到语音: 时长 ${item.duration || 0} 秒，资源定位: ${item.file || item.url}`)
        break
      }
      case 'forward': {
        console.log(`收到聊天记录: ${item.summary}，前置预览条数: ${item.previewMessages?.length || 0}`)
        break
      }
      case 'image': {
        const img = await mediaRefStore.resolveImage(ref, globalThis.onebotClient)
        console.log(`图片 MIME: ${img?.mimeType}，Base64 字符数: ${img?.base64?.length || 0}`)
        break
      }
    }
  })
}

7. 轻量持久化 KV 存储

// 存储数据（自动 JSON 序列化）
await ctx.storage.set('user_level_' + e.userId, 5)

// 获取数据（支持回退默认值）
const level = await ctx.storage.get('user_level_' + e.userId, 1)

await ctx.storage.delete('user_level_' + e.userId)

8. 工具注入 (Function Calling) 与定时任务

export default function setup(ctx) {
  // 1. 注册 Function Calling 工具给预设模型
  ctx.registerTool({
    name: 'query_inventory',
    description: '查询某样物品的当前库存数量',
    parameters: {
      type: 'object',
      properties: {
        itemName: { type: 'string', description: '物品名称' }
      },
      required: ['itemName']
    }
  }, async (args) => {
    return { stock: 42, item: args.itemName }
  })

  // 2. 定时任务（插件重载或禁用时自动安全注销）
  ctx.cron('0 8 * * *', async () => {
    ctx.logger.info('早上 8 点整定时播报触发！')
  })
}
